    id, 
    icn,

    ROW_NUMBER() OVER (PARTITION BY icn ORDER BY create_time DESC) as rn
FROM ietm_icn_manage
WHERE icn = 'ICN-ZB1-A-020000-60101-30101-00017-A-001-01'
ORDER BY create_time DESC;

-- 检查最新ICN的附件（使用pid字段）
SELECT 
    i.id as icn_id,
    i.icn,
    i.create_time as icn_create_time,
    a.id as attachment_id,
    a.pid,           -- 应该等于 icn_id
    a.file_type,     -- 应该是 'entity'
    a.file_key,
    a.file_name
FROM ietm_icn_manage i
LEFT JOIN ietm_attachment a ON a.pid = i.id AND a.file_type = 'entity'
WHERE i.icn = 'ICN-ZB1-A-020000-60101-30101-00017-A-001-01'
ORDER BY i.create_time DESC, a.create_time ASC;
